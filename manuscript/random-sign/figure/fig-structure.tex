% \panel{x-offset}{title}{counts}{key labels}{signs of keys}{signed?}
\newcommand{\panel}[6]{%
  \begin{scope}[xshift=#1]
    \node[anchor=west,font=\small] at (-0.45,2.95) {#2};
    % keys
    \foreach \i/\lab [count=\n] in {#4} {
      \node[circle,draw,thick,minimum size=7.5mm,inner sep=0pt,font=\small]
        (k\n) at ({0.55+(\n-1)*1.55},1.9) {\lab};
    }
    % cells
    \edef\countList{#3}
    \foreach \c [count=\v] in \countList {
      \ifnum\v=1 \def\cellfill{pure}\else\ifnum\v=4 \def\cellfill{pure}\else\ifnum\v=7 \def\cellfill{pure}\else\ifnum\v=8 \def\cellfill{pure}\else\def\cellfill{white}\fi\fi\fi\fi
      \node[draw,thick,fill=\cellfill,minimum width=6.2mm,minimum height=6.2mm,inner sep=0pt,font=\small]
        (c\v) at ({(\v-1)*0.72},0) {$\c$};
      \node[font=\scriptsize,gray] at ({(\v-1)*0.72},-0.48) {\v};
    }
    % edges: S_1={1,2,3}, S_2={3,4,5}, S_3={5,6,7}, S_4={2,6,8}
    \foreach \key/\cells [count=\n] in \mappingEdges {
      \pgfmathsetmacro{\sg}{{#5}[\n-1]}
      \foreach \v in \cells {
        \ifnum#6=1
          \ifdim\sg pt>0pt
            \draw[sgnpos,thick,-] (k\key) -- (c\v.north);
          \else
            \draw[sgnneg,thick,densely dashed,-] (k\key) -- (c\v.north);
          \fi
        \else
          \draw[thick,-] (k\key) -- (c\v.north);
        \fi
      }
    }
  \end{scope}%
}
