\section{Hypergraph lemmas}\label{app:graph}
We prove the hypergraph results used in Theorem~\ref{thm:clt} and the discrete limit distribution: small-core counting, the characterization of failure by a duplicate edge pair when $d=o(M)$, the low-load lemma at fixed subcritical load, and concentration of the overlap graph. These results depend only on the unsigned mapping $\mathcal G$. Here $\mathcal G$ consists of $d$ labeled, independent, uniform $k$-subsets, with duplicate edges allowed.

Fix $k\ge3$. Let $\lambda=\binom d2/\binom Mk$ be the expected number of duplicate edge pairs. Define $Q$ as the planted distribution obtained by choosing a pair of edge labels uniformly and conditioning on the two edges mapping identically. Let $b_k=\lceil3(k-2)/2\rceil$. Write $K=\binom Mk$, let $N=\sum_{i<j}\mathbf1_{\{S_i=S_j\}}$ be the number of duplicate edge pairs, and let $D=\{N\ge1\}$.

\subsection{Small-core counting and duplicate edge pairs}
\begin{lemma}[Small-core counting]\label{lem:smallcore}
A 2-core subgraph with $e$ edges uses at most $v_e=\lfloor ke/2\rfloor$ vertices. For $d\le A_0M$ and $e\le\varepsilon M$ (with $\varepsilon<1/k$), a first-moment bound on the probability that such a subgraph occurs is
\begin{equation*}
B_e=\binom de\binom M{v_e}\Bigl(\binom{v_e}k\Big/\binom Mk\Bigr)^e\le\bigl[C(e/M)^{(k-2)/2}\bigr]^e,
\end{equation*}
where $C$ depends only on $k$ and $A_0$. Retaining the factor in $d$ gives $B_e\le(C_kd/M)^e(e/M)^{\lceil(k-2)e/2\rceil}$.
\end{lemma}

The proof of Theorem 1 in \cite{gm} uses the same first-moment counting argument to obtain a failure probability $O(t^{-k+2})$. Analogous counting arguments for stopping sets appear in \cite{yw,orlitsky2005stopping}. We retain the factor in $d$ for Lemma~\ref{lem:plant}, where $d=o(M)$.

\begin{proof}
Fix a set $I$ of $e$ edge labels and a set $V$ of $v_e$ vertices. The event $A_{I,V}=\{S_i\subseteq V,\ i\in I\}$ includes every core on these edge labels contained in $V$. Summing over $I$ and $V$ gives the first-moment bound $B_e$ in the lemma. Using $\binom de\le(3d/e)^e$, $\binom M{v_e}\le(3M/v_e)^{v_e}$, and $\binom{v_e}k/\binom Mk\le(v_e/M)^k$, together with $ke-v_e=\lceil ke/2\rceil$ and $v_e\le ke/2$, we obtain
\[
B_e\le3^{e+v_e}\Bigl(\frac de\Bigr)^e\Bigl(\frac{v_e}M\Bigr)^{\lceil ke/2\rceil}\le\Bigl(\frac{C_kd}M\Bigr)^e\Bigl(\frac eM\Bigr)^{\lceil(k-2)e/2\rceil},
\]
which retains the factor in $d$. Substituting $d\le A_0M$ and dropping the ceiling, since $e/M<1$, gives the first bound. Let $a=(k-2)/2$. Keeping the ceiling at $e=3$ gives $B_3=O(M^{-b_k})$. For $e\ge4$, the ratio of consecutive terms is at most $Ce^a((e+1)/M)^a$. For fixed $r\ge0$, choose $\varepsilon$ small enough that the ratio of consecutive terms of $e^rB_e$ is at most $1/2$ on $4\le e<\varepsilon M$, so that
\[
\sum_{e=3}^{\lfloor\varepsilon M\rfloor}e^rB_e=O(M^{-b_k}).
\]
This sum also covers the intermediate range with $e\to\infty$ and $e=o(M)$.
\end{proof}


The following lemma quantifies Proposition~\ref{prop:plant}. When $d=o(M)$, failure is almost always caused by a duplicate edge pair, so the failure-conditioned mapping is close to the planted distribution $Q$ in TV. Duplicate edge pairs are the bottleneck of IBLT failure, as identified in \cite{gm}. The density of $Q$ with respect to the original distribution is $N/\lambda$, corresponding to size biasing by $N$ \cite{arratia2019sizebias}.

\begin{lemma}[Failure is determined by a duplicate edge pair when $d=o(M)$]\label{lem:plant}
Let $2\le d=o(M)$, regardless of whether $d\to\infty$. Then
\begin{align*}
\Pr(F\setminus\{\text{there is a duplicate edge pair}\})&\le C_k\Bigl(\frac{d^3}{M^{b_k+3}}+\frac{d^4}{M^{2k}}\Bigr)=o(\lambda),\\
\bigl\|\mathcal L(\mathcal G\mid F)-Q\bigr\|_{TV}&\le C_k\Bigl\{\lambda+dM^{k-b_k-3}+\frac{d^2}{M^k}\Bigr\}\to0.
\end{align*}
\end{lemma}

\begin{proof}
A single edge cannot form a core. Two edges form a core if and only if they map identically. Hence $D\subseteq F$, and the core has at least three edges on $F\setminus D$. Since $d=o(M)$, eventually $d\le\varepsilon M$. Thus every core of size $e\le d$ lies within the range of Lemma~\ref{lem:smallcore}, and we use the bound that retains the factor in $d$. The $e=3$ term is $O_k(d^3M^{-3-b_k})$. For $e\ge4$, we use the envelope without the ceiling, $(C_kd/M)^e(e/M)^{(k-2)e/2}$. Its ratio of consecutive terms is at most $C_k(d/M)((e+1)/M)^{(k-2)/2}$, which tends to zero uniformly over $e\le d=o(M)$. Thus $\sum_{e=4}^dB_e\le C_kd^4/M^{2k}$. Adding the two contributions gives the first bound. Since $\lambda\asymp_kd^2/M^k$ and $b_k\ge k-1$, the ratio of the first bound to $\lambda$ is $O_k(dM^{k-b_k-3}+d^2/M^k)\to0$.

A specified pair of edges maps identically with probability $1/K$, so $\E N=\lambda$. Two distinct edge pairs are simultaneously duplicated with probability $K^{-2}$. If the pairs share an edge, three edges map identically. Otherwise, the two pairs are independent. Hence $\E[N(N-1)]\le\lambda^2$, and $\mathbf1_D\ge N-N(N-1)/2$ gives $\lambda-\lambda^2/2\le\Pr(D)\le\lambda$. The density of $Q$ with respect to the original distribution is exactly $N/\lambda$. By the triangle inequality,
\[
\bigl\|\mathcal L(\mathcal G\mid D)-Q\bigr\|_{TV}\le\frac12\E\Bigl|\frac{\mathbf1_D}{\Pr(D)}-\frac{\mathbf1_D}{\lambda}\Bigr|+\frac1{2\lambda}\E(N-\mathbf1_D)=\frac{\lambda-\Pr(D)}{\lambda}\le\frac\lambda2.
\]
Since $D\subseteq F$, $\|\mathcal L(\mathcal G\mid F)-\mathcal L(\mathcal G\mid D)\|_{TV}=\Pr(F\setminus D)/\Pr(F)\le\Pr(F\setminus D)/\Pr(D)$. Since $\lambda\to0$, eventually $\Pr(D)\ge\lambda/2$. The first bound then makes this term $O_k(dM^{k-b_k-3}+d^2/M^k)$. Adding the two bounds gives the second bound. The proof uses only small-core counting and therefore applies to all $2\le d=o(M)$.
\end{proof}

\subsection{Failure structure at fixed subcritical load}
\begin{lemma}[Low load]\label{lem:propL}
Let $d/M\to\alpha\in(0,\alpha_c)$. Then $\Pr(F\setminus P_2)=O(M^{-b_k})$, $\Pr(F)=\lambda\{1+O(M^{-\delta_k})\}$, and $\|\mathcal L(\mathcal G\mid F)-Q\|_{TV}=O(M^{-\delta_k})$, where $P_2$ denotes the event that the actual core consists of exactly two duplicate edges and $\delta_k=\min\{k-2,\,b_k-(k-2)\}\ge1$.
\end{lemma}

\begin{proof}
We first show that for every fixed $\varepsilon>0$ and $\beta>0$, $\Pr(|\mathrm{core}|\ge\varepsilon M)=o(M^{-\beta})$. This is a variant of a known result. For the binomial hypergraph $H(n,p;k)$, Kim \cite{kim2008poisson}, Theorem 1.7, proves that $\Pr(|\mathrm{core}|\ge\delta n)$ is exponentially small at subcritical load. The same conclusion is stated in \cite{gm}, citing \cite{dn08}, whose model assumes a given bounded degree distribution. Our mapping allows duplicate edges, which occur with polynomially small probability at linear load. We therefore prove the required bound directly using fixed-depth message passing. We follow the finite-round neighborhood analysis and concentration argument for LDPC codes \cite[Theorem~2]{richardson2001capacity}, treating cell degrees separately because they have no fixed upper bound in our model. Let $Z_T$ count the edges whose incoming messages are all $1$ at round $T$. Then $|\mathrm{core}|\le Z_T$ on every graph. Local convergence to a Poisson branching tree gives $\E Z_T/M\to\alpha q_T^k$, where $q_0=1$ and $q_{t+1}=1-\exp(-k\alpha q_t^{k-1})$. When $\alpha<\alpha_c$, this recursion has no positive fixed point, so we can choose $T$ such that $\alpha q_T^k<\varepsilon/4$. On the set $H$ of graphs whose maximum degree does not exceed $\lceil\log M\rceil$, changing $h$ edge coordinates changes $Z_T$ by at most $L_Mh$, where $L_M$ has polylogarithmic order. By the typical bounded differences inequality \cite[Proposition~2]{combes2015mcdiarmid}, $\Pr(Z_T\ge\varepsilon M)\le\Pr(H^c)+\exp\{-cM/L_M^2\}=o(M^{-\beta})$.

On $F\setminus P_2$, the actual core has at least three edges. Lemma~\ref{lem:smallcore} controls cores of size below $\varepsilon M$, and the bound above controls cores of linear size. Together they give $\Pr(F\setminus P_2)=O(M^{-b_k})$. We complete the proof as in the second paragraph of the proof of Lemma~\ref{lem:plant}. The Bonferroni inequality gives $\Pr(F)=\lambda+O(\lambda^2+M^{-b_k})$. We also have $\|\mathcal L(\mathcal G\mid D)-Q\|_{TV}\le\lambda/2$. Since $P_2\subseteq D\subseteq F$, $\|\mathcal L(\mathcal G\mid F)-\mathcal L(\mathcal G\mid D)\|_{TV}\le\Pr(F\setminus P_2)/\Pr(D)$. Using $\lambda\asymp_kM^{2-k}$, the two terms combine to give $O(M^{-\delta_k})$. This total variation bound applies to the entire labeled mapping.
\end{proof}


\subsection{Superpolynomial concentration of the overlap graph}
\begin{lemma}[Superpolynomial graph concentration]\label{lem:mdconc}\label{lem:conc}
Fix $k\ge3$, and let $d/M$ lie in a fixed compact interval of $(0,\infty)$. Write $m_{ij}=H_{ij}$, $\Omega=\sum_{i<j}m_{ij}$, $\Omega_2=\sum_{i<j}m_{ij}^2$, $\ell_i=l_i=\sum_{j\ne i}H_{ij}$, and $\Omega_\ell=\sum_i\ell_i^2$. There is a deterministic set of graphs $\mathcal G_M$ whose complement has probability smaller than any fixed negative power of $M$. On this set, all three statistics deviate from their exact means by at most $O(\sqrt M(\log M)^4)$, and
\begin{equation*}
\Omega_2=\binom d2\left\{\frac{k^2}M+\frac{k^2(k-1)^2}{M(M-1)}\right\}+O(\sqrt M(\log M)^4),\qquad
\max_i l_i\le k\lceil\log M\rceil.
\end{equation*}
\end{lemma}

\begin{proof}
Write $D_v$ for the degree of cell $v$, let $L=\lceil\log M\rceil$ and $\mathcal D_L=\{\max_vD_v\le L\}$. Since $D_v\sim\mathrm{Bin}(d,k/M)$, $\Pr(\mathcal D_L^c)\le M(eC/L)^L$, which is superpolynomially small.

\emph{Lipschitz bound.} For two graphs in $\mathcal D_L$, we successively replace each edge that differs by the edge with the same label in the other graph. The incidence set of every intermediate graph lies in the union of the two endpoint graphs, so every vertex has degree at most $2L$. Replacing one edge affects only the $O(L)$ edge labels overlapping the old or new edge. Since each $m_{ij}\le k$, both $\Omega$ and $\Omega_2$ change by $O(L)$. For the replaced edge, $\ell_i=O(L)$ and its square changes by $O(L^2)$. The $\ell_j$ values for the other $O(L)$ affected labels change by $O(1)$ each, and their squares by $O(L)$. Thus the total change in $\Omega_\ell$ is $O(L^2)$. All three statistics are therefore $CL^2$-Lipschitz on $\mathcal D_L$ with respect to the Hamming distance over edge labels.

\emph{Concentration.} The typical bounded differences inequality \cite{warnke2016typical} gives concentration for functions that are Hamming Lipschitz on the good set. We use the form due to Combes \cite[Proposition~2]{combes2015mcdiarmid}. We take a Lipschitz extension of $f$ from $\mathcal D_L$, truncate it to a deterministic nonnegative polynomial range, and apply McDiarmid's inequality to the independent edges. The bad set has superpolynomially small probability. Both $f$ and its truncated extension are at most $O(M^3)$, so the change in expectation is also superpolynomially small. Hence
\[
\Pr\bigl(|f-\E f|>C\sqrt M(\log M)^4\bigr)\le\Pr(\mathcal D_L^c)+2e^{-c(\log M)^4},\qquad f=\Omega,\Omega_2,\Omega_\ell.
\]

\emph{Deterministic means.} For the overlap $H$ of independent uniform $k$-subsets, write $h_1=\E H=k^2/M$ and $h_2=\E H^2=k^2/M+k^2(k-1)^2/[M(M-1)]$. The exact means are
\[
\E\Omega=\tbinom d2h_1,\qquad\E\Omega_2=\tbinom d2h_2,\qquad\E\Omega_\ell=d\{(d-1)h_2+(d-1)(d-2)h_1^2\}.
\]
The last identity holds because, given $S_i$, the $m_{ij}$ are independent and their conditional first and second moments do not depend on $S_i$. We take the good set to be the intersection of $\mathcal D_L$ with the three deviation events, which gives the stated concentration bound. Since $l_i\le kL$, the lemma follows.
\end{proof}
